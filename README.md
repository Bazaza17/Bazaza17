// Turns a photo into the ASCII portrait shown on the home page.
// Usage: node scripts/portrait.mjs IMG_4166.jpg
// Writes content/portrait.ts. The photo itself is never committed.

import fs from "node:fs";
import sharp from "sharp";

const input = process.argv[2];
if (!input) throw new Error("Usage: node scripts/portrait.mjs <photo>");

// Crop to head and shoulders, in pixels of the original photo.
const CROP = { left: 230, top: 690, width: 820, height: 1000 };
const COLS = 90;
// A monospace character is ~0.6 as wide as it is tall, so use fewer rows to keep proportions.
const ROWS = Math.round(COLS * (CROP.height / CROP.width) * 0.6);
// Characters from empty to dense.
const RAMP = " .:-=+*#%@";
// > 1 pushes mid-tones (skin, wall) toward blank so hair, eyes, and shadows carry the image.
const GAMMA = 2.2;
// Anything fainter than this becomes a space, which clears background texture.
const FLOOR = 0.1;
// Fade toward blank over this fraction of the width/height at each edge, so there's no hard box.
const EDGE = 0.18;

function edgeFade(x, y) {
  const fx = Math.min(x + 0.5, COLS - x - 0.5) / (COLS * EDGE);
  const fy = Math.min(y + 0.5, ROWS - y - 0.5) / (ROWS * EDGE);
  return Math.min(1, fx) * Math.min(1, fy);
}

// Two passes: sharp always resizes before filtering, and the contrast step needs full resolution.
const enhanced = await sharp(input)
  .rotate() // respect the phone's EXIF orientation
  .extract(CROP)
  .grayscale()
  .clahe({ width: 120, height: 120, maxSlope: 2 }) // local contrast, so facial features stand out
  .normalize()
  .toBuffer();

const { data, info } = await sharp(enhanced)
  .resize(COLS, ROWS, { fit: "fill" })
  .toColourspace("b-w")
  .raw()
  .toBuffer({ resolveWithObject: true });
if (info.channels !== 1) throw new Error(`Expected 1 channel, got ${info.channels}`);

const lines = [];
for (let y = 0; y < ROWS; y++) {
  let line = "";
  for (let x = 0; x < COLS; x++) {
    const darkness = 1 - data[y * COLS + x] / 255;
    const ink = Math.max(0, (darkness ** GAMMA * edgeFade(x, y) - FLOOR) / (1 - FLOOR));
    line += RAMP[Math.min(RAMP.length - 1, Math.floor(ink * RAMP.length))];
  }
  lines.push(line.trimEnd());
}

// In dark mode the same text renders light-on-dark, which reads as a glowing negative.
fs.writeFileSync(
  "content/portrait.ts",
  `// Generated by scripts/portrait.mjs. Do not edit by hand.\n\nexport const portrait = ${JSON.stringify(lines.join("\n"))};\n`,
);
console.log(`Wrote content/portrait.ts (${COLS}x${ROWS})`);
